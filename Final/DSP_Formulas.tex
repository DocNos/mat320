\textsc{$R_\omega$: magnitude response }
}
\begin{multicols}{2}

$H(\omega) = 1 - 2e^{-i\omega} = \frac{e^{i\omega} - 2}{e^{i\omega}}$ \\

$\theta_\omega = tan^{-1}(\frac{Imaginary Part}{RealPart})$\\
$e^{i\theta\omega} = \frac{H(\omega)}{|H(\omega)|}$
$\theta_\omega H(\omega) = e^{i\omega}(1 + ae^{-i\omega})$ \\
\end{multicols}

\subsubsection*{Feedback Filters}
\textbf{First Order:} $y_t = x_t + Ry_{t-1}$
\begin{itemize}
\item $H(z) = \frac{1}{1-Rz^{-1}} = \frac{z}{z-R}$
\item Pole at $z = R$ (stable if $|R| < 1$)
\item Impulse response: $h[n] = R^n u[n]$
\item $|H(\omega)| = \frac{1}{|1 - Re^{-i\omega}|}$
\end{itemize}

\textbf{Reson Filter (2-Pole)}\\
\textbf{Poles:} $z = Re^{\pm i\theta}$

\important{H(z) = \frac{1}{1 - 2R\cos\theta \cdot z^{-1} + R^2z^{-2}}}

\textbf{Properties:}
\begin{itemize}
\item Peak at $\omega = \theta$ (resonant freq)
\item Bandwidth: $B \approx 2(1-R)$
\item For narrow band: $R \approx 1 - B/2$
\item Peak gain: $\approx \frac{1}{1-R}$ for $R \approx 1$
\item Max $|H(\omega)|$: $\approx \frac{1}{\sqrt{1-R^2}}$ at resonance
\item True max: $\cos(\omega_m) = \frac{1+R^2}{2R}$
\end{itemize}

\textbf{Implementation:}
$y_t = x_t + 2R\cos\theta \cdot y_{t-1} - R^2y_{t-2}$

\subsection*{Comb Filter}
\textbf{Form:} $H(z) = \frac{1}{1 - R^Lz^{-L}}$

\textbf{Impulse Response Sum:}
$\sum_{t=0}^{\infty} h_t = \frac{1}{1-R^L}$

\textbf{Properties:}
\begin{itemize}
\item Peaks at $\omega = 2\pi k/L$
\item Nulls depend on $R$ value
\item Creates harmonic series
\end{itemize}

\subsection*{All-Pass Filter}
\textbf{General Form:}
$H(z) = \frac{a + z^{-1}}{1 + az^{-1}}$ where $|a| < 1$

\textbf{Properties:}
\begin{itemize}
\item $|H(\omega)| = 1$ for all $\omega$
\item Phase adjustment without magnitude change
\item Numerator = conjugate reciprocal of denominator
\item Used for fractional delay
\end{itemize}

\textbf{Phase Delay:}
$\tau(\omega) = \frac{-2\arctan(\frac{a\sin\omega}{1+a\cos\omega})}{\omega}$

\subsection*{Plucked String (Karplus-Strong)}
\textbf{Components:}
\begin{itemize}
\item Delay line: $L$ samples
\item Low-pass: $y = \frac{1}{2}(x_t + x_{t-1})$, adds 0.5 delay
\item All-pass: fine tuning
\end{itemize}

\textbf{Frequency Tuning:}
\begin{itemize}
\item Period: $P = f_s/f_0$ samples
\item Integer delay: $L = \lfloor P \rfloor$
\item Fractional: $\delta = P - L$
\item All-pass coeff: $a = \frac{1-\delta}{1+\delta}$
\end{itemize}

\subsection*{Square Wave Fourier}
Period $T$, 50\% duty cycle:
$$c_k = \begin{cases}
0 & k \text{ even} \\
\frac{4}{k\pi} & k \text{ odd}
\end{cases}$$

\textbf{General Fourier:}
$c_k = \frac{1}{T}\int_0^T s(t)e^{-i2\pi kt/T}dt$

\subsection*{Filter Combinations}
\begin{itemize}
\item \textbf{Cascade:} $H_{total} = H_1 \cdot H_2$
\item \textbf{Parallel:} $H_{total} = H_1 + H_2$
\item \textbf{Feedback:} $H_{total} = \frac{GH}{1-FH}$
\end{itemize}


\subsection*{Phase Properties}
\textbf{Linear Phase:}
\begin{itemize}
\item Symmetric coefficients
\item $\phi(\omega) = -\omega D$ where $D$ = group delay
\item No distortion of wave shape
\end{itemize}

\textbf{Phase Delay vs Group Delay:}
\begin{itemize}
\item Phase delay: $\tau_p = -\phi(\omega)/\omega$
\item Group delay: $\tau_g = -d\phi/d\omega$
\end{itemize}
